### 获取记录详情

```javascript
fetchRecord(record) {
  if (!record || !record.id || !this.url.queryById) return Promise.resolve(record || {})
  return getAction(this.url.queryById, { id: record.id })
    .then((res) => (res && res.success && res.result ? res.result : record))
    .catch(() => record)
}
```

### 格式化值

```javascript
formatValue(value) {
  return value === undefined || value === null || (typeof value === 'string' && value.trim() === '') ? '-' : value
}
```

### 渲染Tooltip

```javascript
renderTooltip(text) {
  const value = this.formatValue(text)
  return (
    <a-tooltip placement="topLeft" title={value}>
      <span class="table-cell-ellipsis">{value}</span>
    </a-tooltip>
  )
}
```

### 获取序号

```javascript
getRowIndex(index) {
  return (this.ipagination.current - 1) * this.ipagination.pageSize + index + 1
}
```

### 获取状态文本

```javascript
getStatusText(status) {
  return STATUS_MAP[status] ? STATUS_MAP[status].text : this.formatValue(status)
}
```

### 获取状态颜色

```javascript
getStatusColor(status) {
  return STATUS_MAP[status] ? STATUS_MAP[status].color : ''
}
```

### 处理选择值

```javascript
handleSelectValue(field, value, rows) {
  this.queryParam[field] =
    (rows || [])
      .map((item) => item.name)
      .filter(Boolean)
      .join(',') ||
    value ||
    ''
}
```

### 状态切换

```javascript
handleStatusChange() {
  this.queryParam.mszt = this.currentStatus === 'all' ? '' : this.currentStatus
  this.loadData(1)
  this.onClearSelected()
}
```

### 查询

```javascript
searchQuery() {
  this.currentStatus = this.queryParam.mszt || 'all'
  this.loadData(1)
  this.onClearSelected()
}
```

### 重置

```javascript
searchReset() {
  this.queryParam = this.getDefaultQueryParam()
  this.currentStatus = 'all'
  this.loadData(1)
  this.onClearSelected()
}
```

## 字段配置集成

面试管理已集成字段配置功能：

```vue
<template>
  <!-- 字段配置按钮 -->
  <a-button type="primary" class="pm-btn-primary" icon="setting" @click="toggleColumnConfig">
    字段配置
  </a-button>
  
  <!-- 字段配置侧边栏 -->
  <column-config-sidebar
    :visible="columnConfigVisible"
    :allColumns="columnConfigNew"
    :fieldTitleMap="fieldTitleMap"
    @close="columnConfigVisible = false"
    @confirm="handleColumnConfigConfirm"
  />
</template>

<script>
import { columnConfigMixin } from '@/mixins/columnConfigMixin'
import ColumnConfigSidebar from '@comp/common/ColumnConfigSidebar.vue'

export default {
  mixins: [JeecgListMixin, mixinDevice, columnConfigMixin],
  components: { ColumnConfigSidebar },
  data() {
    return {
      columnCode: 'HrmMsgl', // 字段配置组件标识
      originalColumns: [/* 列定义 */]
    }
  },
  computed: {
    fieldTitleMap() {
      const map = {}
      this.originalColumns.forEach(col => {
        if (col.dataIndex && col.title) {
          map[col.dataIndex] = col.title
        }
      })
      return map
    },
    columns() {
      const fixedColumns = [
        { title: '操作', dataIndex: 'action', align: 'center', fixed: 'right', width: 260, scopedSlots: { customRender: 'action' } }
      ]
      return this.applyColumnConfig(this.originalColumns, { fixedColumns })
    },
    columnConfigNew() {
      return this.getColumnConfigData()
    }
  }
}
</script>
```

## 数据流转

### 面试流程

```
简历筛选 → 待面试 → 面试中 → 面试通过 → 已发offer
                        ↓
                      已淘汰
```

### 状态转换

- **待面试 → 面试中**：开始初试
- **面试中 → 面试通过**：面试通过
- **面试中 → 已淘汰**：面试淘汰
- **面试通过 → 已发offer**：发起Offer

### 面试操作

#### 开始初试

- 记录实际面试时间
- 更新面试状态为"面试中"

#### 改期

- 修改预计面试时间
- 通知候选人

#### 安排部门复试

- 添加复试面试官
- 设置复试时间和地点
- 更新面试轮次

#### 发起Offer

- 跳转到Offer管理
- 自动填充候选人信息
